const numbers = [10, 25, 30, 45, 50, 65, 80];

// 1. Map: Transforming each element
const doubled = numbers.map(n => n * 2);

// 2. Filter: Selecting specific elements
const greaterThan50 = doubled.filter(n => n > 50);

// 3. Reduce: Consolidating into a single value
const totalSum = greaterThan50.reduce((acc, curr) => acc + curr, 0);
